<script>
let heroHP = 100;
let monsterHP = 100;
let used = [];

const questions = [
{q:"每天建議睡眠幾小時？",a:["3","5","7-9","12"],c:2},
{q:"哪個是健康飲食？",a:["炸雞","蔬菜","糖果","汽水"],c:1},
{q:"BMI過高代表？",a:["過瘦","正常","過重","身高高"],c:2},
{q:"每天建議喝水量約？",a:["500ml","1000ml","2000ml","5000ml"],c:2},
{q:"心跳正常範圍？",a:["20","60-100","150","200"],c:1},
{q:"哪種是有氧運動？",a:["舉重","跑步","發呆","滑手機"],c:1},
{q:"維生素C來源？",a:["橘子","牛肉","米飯","奶油"],c:0},
{q:"長期熬夜會？",a:["變健康","免疫下降","變高","無影響"],c:1},
{q:"哪個是蛋白質來源？",a:["雞蛋","糖","鹽","油"],c:0},
{q:"過量糖分會？",a:["變瘦","蛀牙","變聰明","無影響"],c:1}
];

for(let i=questions.length;i<100;i++){
questions.push({q:"健康知識題 "+(i+1),a:["A","B","C","D"],c:Math.floor(Math.random()*4)});
}

function getQ(){
if(used.length===100){alert("全破！");location.reload();}
let i;
do{i=Math.floor(Math.random()*100);}while(used.includes(i));
used.push(i);
return questions[i];
}

function updateHP(){
document.querySelector('#heroBar .hp').style.width=heroHP+'%';
document.querySelector('#monsterBar .hp').style.width=monsterHP+'%';
}

function attackHero(){
heroHP-=10;
document.getElementById('hero').style.transform='translateX(20px)';
setTimeout(()=>{document.getElementById('hero').style.transform='';},200);
document.getElementById('hurt').play();
}

function attackMonster(){
monsterHP-=10;
document.getElementById('monster').style.transform='translateX(-20px)';
setTimeout(()=>{document.getElementById('monster').style.transform='';},200);
document.getElementById('hit').play();
}

function loadQ(){
const q=getQ();
document.getElementById('question').innerText=q.q;
const cdiv=document.getElementById('choices');
cdiv.innerHTML='';
q.a.forEach((opt,i)=>{
let b=document.createElement('button');
b.innerText=opt;
b.onclick=()=>{
if(i===q.c){attackMonster();}
else{attackHero();}
updateHP();
check();
loadQ();
};
cdiv.appendChild(b);
});
}

function check(){
if(heroHP<=0){alert('你輸了');location.reload();}
if(monsterHP<=0){alert('你贏了');location.reload();}
}

loadQ();
</script>
